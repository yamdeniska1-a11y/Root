local SoundService = game:GetService("SoundService")

local BASE_URL     = "https://raw.githubusercontent.com/yamdeniska1-a11y/Root/main/"
local ROOT_FOLDER  = "RootLib"
local SOUND_FOLDER = "RootLib/Sound"

local SoundController = {
    ScrollEnabled = true,
    ClickEnabled = true,
    NotificationVolume = 1,
    ScrollVolume = 0.2,
    ClickVolume = 0.2,
    LastScrollPlay = 0,
    Ready = false,
    Initialized = false,
    SoundFolder = SOUND_FOLDER,
    Assets = {
        Scroll = {
            Url = BASE_URL .. "Scroll.ogg",
            File = SOUND_FOLDER .. "/Scroll.ogg",
            SoundId = nil,
            Instance = nil
        },
        Click = {
            Url = BASE_URL .. "Click.ogg",
            File = SOUND_FOLDER .. "/Click.ogg",
            SoundId = nil,
            Instance = nil
        },
        Notification = {
            Url = BASE_URL .. "Notification.ogg",
            File = SOUND_FOLDER .. "/Notification.ogg",
            SoundId = nil,
            Instance = nil
        }
    }
}

local function ensureFolders()
    pcall(function()
        if makefolder and isfolder then
            if not isfolder(ROOT_FOLDER) then makefolder(ROOT_FOLDER) end
            if not isfolder(SOUND_FOLDER) then makefolder(SOUND_FOLDER) end
        end
    end)
end

local function isValidData(data)
    return type(data) == "string" and #data > 64 and data:sub(1, 3) ~= "404"
end

local function cleanupOldSounds()
    pcall(function()
        for _, child in ipairs(SoundService:GetChildren()) do
            if child:IsA("Sound") and child.Name:sub(1, 13) == "RootLibSound_" then
                child:Destroy()
            end
        end
    end)
end

function SoundController:Init()
    if self.Initialized then return end
    self.Initialized = true

    task.spawn(function()
        ensureFolders()
        cleanupOldSounds()

        local getAsset = getcustomasset or getsynasset

        for key, info in pairs(self.Assets) do
            pcall(function()
                if writefile and isfile and getAsset then
                   local needDownload = not isfile(info.File)
                    if not needDownload and readfile then
                        local okRead, content = pcall(readfile, info.File)
                        needDownload = (not okRead) or (#content < 64)
                    end
                    if needDownload then
                        local ok, data = pcall(function()
                            return game:HttpGet(info.Url)
                        end)
                        if ok and isValidData(data) then
                            writefile(info.File, data)
                        end
                    end
                    if isfile(info.File) then
                        info.SoundId = getAsset(info.File)
                    end
                end

                local snd = Instance.new("Sound")
                snd.Name = "RootLibSound_" .. key
                if info.SoundId then
                    snd.SoundId = info.SoundId
                end
                snd.Volume = (key == "Notification") and self.NotificationVolume or 0.2
                snd.Parent = SoundService
                info.Instance = snd
            end)
        end

        self.Ready = true
    end)
end

local function playAsset(self, key, volume)
    pcall(function()
        local info = self.Assets[key]
        local snd = info and info.Instance
        if snd and snd.SoundId and snd.SoundId ~= "" then
            snd.Volume = volume
            snd.TimePosition = 0
            snd:Play()
        end
    end)
end

function SoundController:PlayScroll()
    if not self.ScrollEnabled then return end
    local now = tick()
    if now - self.LastScrollPlay < 0.035 then return end
    self.LastScrollPlay = now
    playAsset(self, "Scroll", self.ScrollVolume)
end

function SoundController:PlayClick()
    if not self.ClickEnabled then return end
    playAsset(self, "Click", self.ClickVolume)
end

function SoundController:PlayNotification()
    if self.NotificationVolume <= 0 then return end
    playAsset(self, "Notification", self.NotificationVolume)
end

function SoundController:Register(key, url, volume)
    if self.Assets[key] then return end
    self.Assets[key] = {
        Url = url,
        File = SOUND_FOLDER .. "/" .. key .. ".ogg",
        SoundId = nil,
        Instance = nil
    }
    self.Initialized = false
    self:Init()
end

function SoundController:Play(key, volume)
    playAsset(self, key, volume or 0.5)
end

function SoundController:Destroy()
    for _, info in pairs(self.Assets) do
        pcall(function()
            if info.Instance then info.Instance:Destroy() end
        end)
        info.Instance = nil
    end
    self.Ready = false
    self.Initialized = false
end

return SoundController
